### LratCheck.lean
```lean
import Std.Tactic.BVDecide.LRAT
open Std.Sat Std.Tactic.BVDecide

namespace LratCheck

/-- Parse a DIMACS CNF text. DIMACS variable `v ≥ 1` becomes atom `v - 1 : Nat` (the convention of `CNF.dimacs` and `LRAT.check`);
literal `+v` is `(v-1, true)`, `-v` is `(v-1, false)`. Comment lines (`c`) and the header (`p`) are skipped.
A clause is terminated by `0`. Returns the clauses in file order. -/
def parseDimacs (s : String) : CNF Nat := Id.run do
  let b := s.toUTF8
  let n := b.size
  let mut i := 0
  let mut clauses : Array (CNF.Clause Nat) := #[]
  let mut cur : CNF.Clause Nat := CNF.Clause.empty
  let mut lineStart := true
  while i < n do
    let c := b.get! i
    if lineStart && (c == 'c'.toUInt8 || c == 'p'.toUInt8) then
      -- skip the line
      while i < n && b.get! i != 10 do i := i + 1
      i := i + 1
      lineStart := true
    else if c == 10 then
      i := i + 1; lineStart := true
    else if c == 32 || c == 9 || c == 13 then
      i := i + 1
    else
      lineStart := false
      let mut neg := false
      if c == '-'.toUInt8 then neg := true; i := i + 1
      let mut v : Nat := 0
      while i < n && b.get! i ≥ 48 && b.get! i ≤ 57 do
        v := v * 10 + (b.get! i - 48).toNat
        i := i + 1
      if v == 0 then
        clauses := clauses.push cur
        cur := CNF.Clause.empty
      else
        cur := cur.add (v - 1) (!neg)
  return ⟨clauses⟩

def lratOf (s : String) : Array LRAT.IntAction :=
  match LRAT.parseLRATProof s.toUTF8 with
  | .ok p => p
  | .error _ => #[]

end LratCheck
```

### renum.py (untrusted: drat-trim LRAT ids are not contiguous; Lean core's checker indexes clauses by position)
```python
#!/usr/bin/env python3
"""renum.py NCLAUSES IN.lrat OUT.lrat : renumber LRAT lemma ids to be contiguous (NCLAUSES+1, +2, ...),
as Lean core's LRAT checker indexes clauses by position. Untrusted preprocessing (a wrong renumbering can only make
the verified checker reject)."""
import sys
m=int(sys.argv[1]); nxt=m+1; mp={}
def f(t):
    v=int(t)
    if v<0: return '-'+str(mp.get(-v,-v))
    return str(mp.get(v,v))
with open(sys.argv[2]) as fi, open(sys.argv[3],'w') as fo:
    for line in fi:
        t=line.split()
        if not t: continue
        if t[1]=='d':
            fo.write(str(nxt-1)+' d '+' '.join(f(x) for x in t[2:])+'\n')
            continue
        i=int(t[0]); z=t.index('0',1)
        lits=t[1:z]; hints=t[z+1:]
        if lits:  # non-empty lemma gets a fresh contiguous id
            mp[i]=nxt; nid=nxt; nxt+=1
        else:
            nid=nxt
        fo.write(str(nid)+' '+' '.join(lits+['0']+[f(x) for x in hints])+'\n')
```
